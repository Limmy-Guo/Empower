    function renderEvents(f){
      const list=stage.querySelector('#events-list');list.innerHTML="";
      let filtered=eventsData.filter(e=>(!f.time||e.time===f.time)&&(!f.type||e.type===f.type)&&(!f.location||e.location===f.location));
      stage.querySelector('#filter-result').innerText=filtered.length?`Showing ${filtered.length} events`:"No events found";
      filtered.forEach(e=>{
        list.innerHTML+=`<div class="card"><div class="icon" style="background:${e.color};">${e.icon}</div>
          <div class="text"><h4>${e.title}</h4><p>${e.desc}</p><div class="meta">${e.time} · ${e.type} · ${e.location}</div></div></div>`;
      });
    }

    function initEvents(){

      ["time","type","location"].forEach(k=>stage.querySelector(`#filter-${k}`).addEventListener('change',()=>{
        renderEvents({time:stage.querySelector('#filter-time').value,type:stage.querySelector('#filter-type').value,location:stage.querySelector('#filter-location').value});
      }));
      setupNavigation();
    }
